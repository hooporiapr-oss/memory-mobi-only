-- Adds school_name to the existing Handle standings table. -- -- This table was originally built with no school concept at all — -- one single, global board, since the mobile gate only ever checked -- "is this code valid," not which school it belonged to. Now that -- the gate captures and stores the school name from the license at -- login, the game can tag every saved score with it, and the -- standings view can filter by it — so schools sharing this same -- mobile site each see their own board instead of one merged one. -- -- This is a genuine ALTER on the live table, not a fresh CREATE — -- existing rows are untouched, they'll just show a blank school_name -- until new games are played and saved with one attached. -- -- Run this once in your Supabase project's SQL Editor. alter table ritnome_mobile_handle_4qtr_standings add column if not exists school_name text; create index if not exists ritnome_mobile_handle_4qtr_standings_school_idx on ritnome_mobile_handle_4qtr_standings (school_name); -- Confirms the column is genuinely there now. select column_name, data_type from information_schema.columns where table_name = 'ritnome_mobile_handle_4qtr_standings' order by ordinal_position;
